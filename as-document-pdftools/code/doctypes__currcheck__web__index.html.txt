function findCodes(t) {
  t = thaiDigits(t);
  var out = [], m;
  var en = /(^|[^A-Za-z])([A-Z]{3,4})\s*(\d{3})(?!\d)/g;
  while ((m = en.exec(t))) out.push({ script: 'en', prefix: m[2], num: m[3], idx: m.index + m[1].length, len: m[0].length - m[1].length });
  var th = /(^|[\s*(\[])([ก-ฮ]{2,5})\s*(\d{3})(?![\d\/.-])/g;
  while ((m = th.exec(t))) out.push({ script: 'th', prefix: m[2], num: m[3], idx: m.index + m[1].length, len: m[0].length - m[1].length });

}

// แยกบรรทัดพร้อมเลขหน้า (ข้อความจาก pdfToText มีบรรทัด @@PAGE n@@ คั่นหน้า)
function splitPages(text) {
  var page = 1, out = [];
  String(text || '').split(/\r?\n/).forEach(function (raw) {
    var m = raw.match(/^@@PAGE (\d+)@@$/);
    if (m) { page = +m[1]; return; }
    out.push({ raw: raw.trim(), t: thaiDigits(raw).trim(), page: page });
  });
  return out;
}
function normalizePlan(value) {
  var rawPlan = thaiDigits(value || '');
  if (/(?:แผน|Plan)?\s*1\.2\b/i.test(rawPlan)) return 'ก2';
  if (/(?:แผน|Plan)?\s*1\.1\b/i.test(rawPlan)) return 'ก1';
  if (/(?:แผน|Plan)\s*2\s*(?:แบบ)?(?:วิชาชีพ|Professional)/i.test(rawPlan)) return 'ข';
  var t = thaiDigits(value || '').replace(/\s+/g, '').replace(/แผนการศึกษา|แผน|PLAN/gi, '').toUpperCase();
  if (/^(?:ก(?:แบบก)?|A)\(?1\)?$/.test(t) || t === '1' || t === 'A(A1)') return 'ก1';
  if (/^(?:ก(?:แบบก)?|A)\(?2\)?$/.test(t) || t === '2' || t === 'A(A2)') return 'ก2';
  if (/^(?:ข|B)$/.test(t)) return 'ข';
  return t;
}

function cleanPdfText(s) {
  return String(s || '').normalize('NFC').replace(/[\u200B\uFEFF]/g, '')
    .replace(/\uF710/g, 'ั').replace(/\uF70E/g, '์').replace(/\uF70A/g, '่').replace(/\uF70B/g, '้').replace(/\uF70C/g, '๊').replace(/\uF70D/g, '๋')
    .replace(/([ก-๙])\s+([ัิีึืุู็่้๊๋์])/g, '$1$2');
}
// Repair only known Thai structural keywords from legacy PDF font encodings.
function normalizeBookKeywords(text) {
  return String(text).replace(/หน[PGH]วยกิต/g, 'หน่วยกิต')
    .replace(/โครงสร[NQ]าง(?:ของ)?หลักสูตร/g, 'โครงสร้างหลักสูตร')
    .replace(/ไม[PG]น[$N]อยกว[PG]า/g, 'ไม่น้อยกว่า')
    .replace(/วิทยานิพนธ[?W]/g, 'วิทยานิพนธ์')
    .replace(/การค[$N]นคว[$N]าอิสระ/g, 'การค้นคว้าอิสระ');
}
// Rejoin code columns only when a prefix is immediately followed by its number.
function prepareBookText(text) {
  var lines = normalizeBookKeywords(cleanPdfText(text)).split(/\r?\n/);
  for (var i = 0; i < lines.length; i++) {
    var a = lines[i].trim().replace(/^\*+\s*/, '');
    if (/^(?:[A-Z]{3,4}|[ก-ฮ]{2,5})$/.test(a)) {
      var j = i + 1;
      while (j < lines.length && !lines[j].trim()) j++;
      var next = (lines[j] || '').trim().replace(/^\*+\s*/, '');
      var k = j + 1; while (k < lines.length && !lines[k].trim()) k++;
      var h = k + 1; while (h < lines.length && !lines[h].trim()) h++;
      if (/^[ก-ฮ]{2,5}$/.test(a) && /^[A-Z]{3,4}$/.test(next) && /^[๐-๙0-9]{3}$/.test((lines[k] || '').trim()) && /^[๐-๙0-9]{3}$/.test((lines[h] || '').trim())) {
        lines[i] = a + ' ' + lines[k].trim(); lines[j] = next + ' ' + lines[h].trim(); lines[k] = ''; lines[h] = '';
      } else if (/^[๐-๙0-9]{3}$/.test(next)) { lines[i] = a + ' ' + next; lines[j] = ''; }
    }
  }
  return lines.filter(function(x) { return x.trim(); }).join('\n');
}

      res.choiceSets = res.choiceSets || [];
      res.choiceSets.push({group:cur.label,codes:codes.map(function(c) {return c.prefix+c.num;}),count:1,evidence:{page:L[i].page,line:L[i].raw}});
      continue;
    }
    // Mentions in explanatory paragraphs are not course definitions.
    if (codes.length && !/^[\s*]*$/.test(line.slice(0,codes[0].idx))) continue;
    var g2 = codes.length ? null : groupFromLine(line);
    if (g2 && res.courses.length && /^(?:Thesis|Independent Study|Thematic Paper)$/i.test(line.trim()) && i - res.courses[res.courses.length-1].lineNo <= 2) g2 = null;
    if (g2) { major = ''; cur = g2; curHead = { page: L[i].page, line: L[i].raw }; lastCredit = null; addGroup(g2); continue; }
    var cr = line.match(CREDIT_RE);
    if (cr && !codes.length) { var prev = res.courses[res.courses.length - 1]; if (prev && prev.group === (cur && cur.label) && prev.credits == null) { prev.credits = +cr[1]; prev.evidence.lines.push(L[i].raw); } else { lastCredit = +cr[1]; lastCreditLine = L[i].raw; } }
    if (!cur) continue;
    codes.forEach(function (c, ci) {
      var key = c.prefix + c.num;
      var rest = line.slice(c.idx + c.len, ci + 1 < codes.length ? codes[ci + 1].idx : undefined);
      var name = rest.replace(CREDIT_RE, '').trim();
      var last = res.courses[res.courses.length - 1];
      if (last && last.num === c.num && last.group === cur.label && last.major === major && i - last.lineNo <= 5 && !last[c.script === 'en' ? 'codeEn' : 'codeTh']) {
        if (c.script === 'en') { last.codeEn = key; last.prefixEn = c.prefix; last.nameEn = name; last.code = key; }
        else { last.codeTh = key; last.prefixTh = c.prefix; last.nameTh = name; }
        if (last.credits == null && cr) last.credits = +cr[1];
        last.evidence.lines.push(L[i].raw); last.lineNo = i;
        seen[key] = 1;
        return;
      }
      if (seen[key]) return;
      seen[key] = 1;
      var b = { key: key, code: key, num: c.num, codeEn: '', codeTh: '', prefixEn: '', prefixTh: '', nameEn: '', nameTh: '',
        credits: cr ? +cr[1] : lastCredit, major: major, plan: coursePlan, group: cur.label, lineNo: i,
        evidence: { page: L[i].page, heading: curHead, lines: (cr || !lastCreditLine ? [] : [lastCreditLine]).concat([L[i].raw]) } };
      if (c.script === 'en') { b.codeEn = key; b.prefixEn = c.prefix; b.nameEn = name; }
      else { b.codeTh = key; b.prefixTh = c.prefix; b.nameTh = name; }
      res.courses.push(b);
      lastCredit = null; lastCreditLine = null;
    });
  }
  res.courses.forEach(function (b) {
    b.key = b.codeEn || b.codeTh; b.code = b.key;
    b.display = b.codeEn ? b.codeEn.replace(/^([A-Z]+)/, '$1 ') : b.codeTh.replace(/^([ก-ฮ]+)/, '$1 ');
    b.name = b.nameEn || b.nameTh;
    var p = b.prefixEn || b.prefixTh; res.prefixes[p] = (res.prefixes[p] || 0) + 1;
    delete b.lineNo;
  });

  var majorPrefixes = {};
  res.courses.forEach(function(b) { if (!b.major || !res.groups.some(function(g) {return g.label === b.group && g.cat === 'บังคับ';})) return;
    var list = majorPrefixes[b.prefixEn] = majorPrefixes[b.prefixEn] || []; if (list.indexOf(b.major)<0) list.push(b.major);
  });
  res.courses.forEach(function(b) { if (!b.major && res.groups.some(function(g) {return g.label === b.group && THESIS_CATS.indexOf(g.cat)>=0;}) && (majorPrefixes[b.prefixEn] || []).length===1) b.major = majorPrefixes[b.prefixEn][0]; });
  // แผนเริ่มต้น / ข้อมูลประกอบ
  if (!res.planOrder.length && res.groups.length) { res.plans['default'] = []; res.planOrder.push('default'); }
  var mPlan = all.match(/แผน\s*ก\s*แบบ\s*ก\s*(\d)/);
  res.plan = res.planOrder.filter(function (p) { return p !== 'default'; }).join(', ') || (mPlan ? 'ก' + mPlan[1] : '');
  var trackMinimum = normTh(all).match(/เลอกเรยนรายวชาในกลมความเชยวชาญทสนใจไมนอยกวา(\d+)หนวยกต/);
  if (trackMinimum) res.trackElectiveMinimum = +trackMinimum[1];
  res.diag = { chars: all.replace(/\s/g, '').length, lines: L.length, pages: L.length ? L[L.length - 1].page : 0,
    codes: L.reduce(function (a, l) { return a + findCodes(l.t).length; }, 0),
    headings: L.filter(function (l) { return groupFromLine(l.t); }).length, struct: iStruct >= 0, courseHead: iCourses >= 0 };
  return res;
}

// หมวดและเกณฑ์ที่ใช้กับนักศึกษา (ตามแผนการศึกษา)
function groupsForPlan(curr, plan) {
  var wanted = normalizePlan(plan), keys = curr.planOrder || [], named = keys.filter(function(k) { return k !== 'default'; });
  var pk = curr.plans && curr.plans[wanted] ? wanted : named.length === 0 ? 'default' : (!wanted && named.length === 1 ? named[0] : null);
  var req = pk && curr.plans && curr.plans[pk] || [];
  var unmatched = pk == null;
  return { plan: pk === 'default' ? wanted : pk || wanted, planName: curr.planNames && curr.planNames[pk] || '', unmatched: unmatched,
    total: (curr.planTotals && curr.planTotals[pk]) || curr.totalRequired,
    groups: curr.groups.filter(function(g) { return !req.length || req.some(function(r) {return r.label === g.label;}); }).map(function(g) {
      var r = req.filter(function(r) { return r.label === g.label; })[0] || {};
      return {label:g.label,cat:g.cat,required:r.required == null ? null : r.required,atLeast:!!r.atLeast,evidence:r.evidence || null};
    }) };
}
function curriculumMajors(curr) {
  var out = []; (curr.courses || []).forEach(function(b) { if (b.major && out.indexOf(b.major) < 0) out.push(b.major); });
  return out;
}
function selectMajor(curr, tr) {
  var majors = [];
  curr.courses.forEach(function(b) { if (b.major && majors.indexOf(b.major) < 0) majors.push(b.major); });
  if (majors.length < 2) return {major:majors[0] || '',ambiguous:false};
  var codes = {}; tr.rows.forEach(function(r) { codes[normCode(r.code)] = true; });
  var thesis = curr.courses.filter(function(b) { return b.major && (codes[b.codeEn] || codes[b.codeTh]) && curr.groups.some(function(g) { return g.label === b.group && THESIS_CATS.indexOf(g.cat) >= 0; }); });
  var unique = []; thesis.forEach(function(b) {if(unique.indexOf(b.major)<0) unique.push(b.major);});
  if (unique.length === 1) return {major:unique[0],ambiguous:false,source:'รหัสวิทยานิพนธ์ที่ตรงกับเล่ม'};
  if (unique.length > 1) return {major:'',ambiguous:true};
  var scores = majors.map(function(m) { return {major:m,score:curr.courses.filter(function(b) {return b.major === m && (codes[b.codeEn] || codes[b.codeTh]) && curr.groups.some(function(g) {return g.label === b.group && g.cat === 'บังคับ';});}).length}; }).sort(function(a,b){return b.score-a.score;});
  return scores[0].score > 0 && scores[0].score > scores[1].score ? {major:scores[0].major,ambiguous:false,source:'รายวิชาบังคับที่ตรงกับเล่ม'} : {major:'',ambiguous:true};
}

function needsMajor(curr, gp) { return curr.courses.some(function(b) { return b.major && gp.groups.some(function(g) { return g.label === b.group && g.cat === 'บังคับ' && g.required !== 0; }); }); }
function courseForPlan(b, gp) {
  var g = gp.groups.filter(function(g) { return g.label === b.group; })[0];
  if (!g || g.required === 0 || (b.plan && gp.plan && b.plan !== gp.plan)) return false;
  if (b.major && g.cat !== 'เลือก' && gp.major && b.major !== gp.major) return false;
  if (THESIS_CATS.indexOf(g.cat) >= 0 && g.required != null) return b.credits === g.required;
  return true;
}

// จับคู่รหัสวิชาของนักศึกษากับเล่ม: ตรงทั้งรหัส หรือเลข 3 หลักตรงกัน (531 ↔ 531)
function buildMatcher(curr, stu) {
  var exact = {};
  curr.courses.forEach(function(b) { [b.codeEn, b.codeTh].filter(Boolean).forEach(function(code) { (exact[normCode(code)] = exact[normCode(code)] || []).push(b); }); });
  return function(code) {
    var candidates = exact[normCode(code)] || [];
    if (!candidates.length) return null;
    return { b: candidates[0], how: candidates.length === 1 ? 'exact' : 'ambiguous', candidates: candidates };
  };
}


/** อ่านไฟล์โครงสร้างหลักสูตร/ผลการเรียนของนักศึกษา */
function parseTranscript(text) {
  var L = splitPages(cleanPdfText(text)).filter(function(l) { return l.t; });
  var out = { studentId: '', programKey: '', name: '', program: '', programEn: '', plan: '', criteria: '', rows: [], warnings: [] };
  function field(re) {
    for (var i = 0; i < L.length; i++) {
      var m = L[i].t.match(re); if (!m) continue;
      var value = L[i].t.slice(m[0].length).trim();
      if (!value && L[i+1]) value = L[i+1].t;
      return value;
    }
    return '';
  }
  var id = field(/^รหัส\s*นักศึกษา\s*[:：]?\s*/).match(/^(\d{7,12})(?:\s+([A-Z]{3,6}\/\w+))?/);
  if (id) { out.studentId = id[1]; out.programKey = id[2] || ''; }
  out.name = field(/^ชื่อ\s*[-–—]?\s*สกุล\s*[:：]?\s*/);
  out.program = field(/^สาขาวิชา\s*[:：]?\s*/);
  out.programEn = (L.map(function(l) {return l.t;}).filter(function(t) { return /^MASTER OF|^DOCTOR OF PHILOSOPHY/i.test(t); })[0] || '');
  out.plan = normalizePlan(field(/^(?:แผนการศึกษา|Study Plan|Plan)\s*[:：]?\s*/));
  out.criteria = (field(/^เกณฑ์?\s*[:：]?\s*/).match(/\d{4}/) || [''])[0];
  for (var i = 0; i < L.length; i++) {
    var term = L[i].t.match(/^(\d{4}\/\d)\b/); if (!term) continue;
    var block = L[i].t.slice(term[0].length).trim(), page = L[i].page, j = i+1;
    while (j < L.length && !/^\d{4}\/\d\b/.test(L[j].t) && !/^โครงสร้างหลักสูตร|^ข้อม[ูุ]ลการลงทะเบียน|^Invoice|^\(index-|^\(login/.test(L[j].t)) { block += '\n' + L[j].t; j++; }
    var codes = findCodes(block); if (!codes.length) continue;
    var code = codes[0], rest = block.slice(code.idx + code.len).replace(/เลือก\s*หมวดวิชา/g, '').trim();
    var m = rest.match(/(?:^|[ \t])(\d+(?:\.\d+)?)[ \t]*(A|B\+?|C\+?|D\+?|F|S|U|P|W|IP|I|AU|X|TR|TC|T|O)[ \t]*$/m) || rest.match(/(?:^|\s)(\d+(?:\.\d+)?)(?:[ \t]*\n[ \t]*(A|B\+?|C\+?|D\+?|F|S|U|P|W|IP|I|AU|X|TR|TC|T|O))?[ \t]*$/m);
    if (!m) { out.warnings.push('อ่านหน่วยกิต/เกรดไม่ได้: ' + code.prefix + code.num); continue; }
    out.rows.push({ term: term[1], code: normCode(code.prefix + code.num), title: rest.slice(0,m.index).replace(/\s+/g,' ').trim(), credits: +m[1], grade: m[2] || '', page: page });
    i = j-1;
  }
  if (!out.studentId) out.warnings.push('ไม่พบรหัสนักศึกษา กรุณากรอกยืนยัน');
  if (!out.name || /[\uE000-\uF8FF\uFFFD]/.test(out.name)) out.warnings.push('กรุณาตรวจชื่อกับต้นฉบับ');
  return out;
}


/** ตรวจว่าไฟล์นักศึกษาตรงกับเล่มหลักสูตรหรือไม่ (ก่อนเริ่ม) */
function matchCheck(curr, tr) {
  var seen = {}, stu = [];
  tr.rows.forEach(function (r) { if (!seen[r.code]) { seen[r.code] = 1; stu.push(r); } });
  var match = buildMatcher(curr, stu);
  var list = stu.filter(function (r) { return !/^GRID/.test(r.code) || match(r.code); });
  var hits = list.map(function (r) { return match(r.code); });
  var found = hits.filter(Boolean).length, byNum = hits.filter(function (h) { return h && h.how !== 'exact'; }).length;
  var amb = hits.filter(function (h) { return h && h.how === 'ambiguous'; }).length;
  var checks = [];
  checks.push({ ok: list.length ? found / list.length >= 0.6 : false, label: 'รหัสวิชาตรงกับเล่ม',
    detail: 'พบในเล่ม ' + found + ' จาก ' + list.length + ' รหัส' + (byNum ? ' (จับคู่ด้วยเลขรหัส ' + byNum + ')' : '') + ' · ไม่นับวิชาภาษาอังกฤษ GRID' });
  if (amb) checks.push({ ok: false, label: 'เลขรหัสซ้ำในเล่ม', detail: amb + ' วิชามีเลขเดียวกันหลายรายการในเล่ม — ตรวจหมวดในขั้นที่ 2' });
  var gp = groupsForPlan(curr, tr.plan);
  if (gp.unmatched) checks.push({ok:false,label:'ไม่พบเกณฑ์ของแผนนี้',detail:'โปรดตรวจแผนของนักศึกษาและเล่มหลักสูตร'});
  if (normalizePlan(tr.plan) && curr.planOrder && curr.planOrder.filter(function (p) { return p !== 'default'; }).length)
    checks.push({ ok: !!curr.plans[normalizePlan(tr.plan)], label: 'แผนการศึกษา', detail: 'นักศึกษา ' + tr.plan + ' · เล่มมี ' + curr.plan + (curr.plans[normalizePlan(tr.plan)] ? '' : ' — ใช้เกณฑ์แผน ' + (gp.plan || 'แรก')) });
  return { checks: checks, ok: checks.every(function (c) { return c.ok; }) };
}

/**
 * ตรวจโครงสร้าง
 * opts.overrides = { code: cat }  เจ้าหน้าที่เปลี่ยนหมวด
 * opts.transfer  = { code: true } วิชาในเล่มที่เทียบโอน/ยกเว้น
 */
function evaluate(curr, tr, opts) {
  opts = opts || {};
  var overrides = opts.overrides || {}, transfer = opts.transfer || {};
  var gp = groupsForPlan(curr, tr.plan), majorInfo = opts.major && curriculumMajors(curr).indexOf(opts.major) >= 0 ? {major:opts.major,ambiguous:false,source:'วิชาเอกที่ระบุในขั้นแนบไฟล์'} : selectMajor(curr, tr);
  gp.major = majorInfo.major;
  var groupOf = {}; gp.groups.forEach(function (g) { groupOf[g.label] = g; });

  // รวมรายการลงทะเบียนตามรหัสวิชา
  var by = {}, order = [];
  tr.rows.forEach(function (r) {
    var e = by[r.code];
    if (!e) { e = by[r.code] = { code: r.code, title: r.title, regs: [], maxCr: 0 }; order.push(r.code); }
    e.regs.push(r); e.maxCr = Math.max(e.maxCr, r.credits);
  });

  var match = buildMatcher(curr, order.map(function (c) { return { code: c, credits: by[c].maxCr }; }));
  var courses = order.map(function (code) {
    var e = by[code], hit = match(code), b = hit ? hit.b : null;
    var passRegs = e.regs.filter(function (r) { return isPass(r.grade); });
    var pass = passRegs.length > 0;
    var pending = !pass && e.regs.some(function (r) { return isPending(r.grade) || FAIL_GRADES.indexOf(r.grade) < 0; });
    var main = pass ? passRegs[0] : e.regs[0]; // แถวบนสุด = ภาคล่าสุด
    var g = b ? groupOf[b.group] : null;
    var proposed, reason;
    var thesisLike = /^(?:THESIS|THEMATIC PAPER|DISSERTATION|INDEPENDENT STUDY)$/i.test(e.title.trim());
    if (b && g) {
      proposed = g.cat;
      reason = 'พบในเล่ม: ' + g.label + (hit.how === 'exact' ? '' : ' · จับคู่ด้วยเลข ' + b.num + ' กับ ' + b.display);
      if (hit.how === 'ambiguous') reason += ' · เลข ' + b.num + ' มี ' + hit.candidates.length + ' รายการในเล่ม (' + hit.candidates.map(function (c) { return c.display + ' ' + c.group; }).join(', ') + ') — โปรดตรวจ';
      if (THESIS_CATS.indexOf(g.cat) >= 0) reason += ' · ลงทะเบียน ' + e.maxCr + ' หน่วยกิต → ' + proposed;
    } else if (/^GRID/.test(code) && /ENGLISH|ภาษาอังกฤษ|GRAMMAR|(?:READ|SPEAK).*ACADE.*CONT.*GR/i.test(e.title)) {
      proposed = 'ภาษาอังกฤษ'; reason = 'รหัส GRID ที่ไม่มีในเล่ม = วิชาภาษาอังกฤษของบัณฑิตวิทยาลัย';
    } else if (thesisLike) {
      proposed = thesisCatByCredits(e.maxCr); reason = 'ไม่มีในเล่ม แต่ชื่อวิชาเป็นวิทยานิพนธ์/สารนิพนธ์ · ลงทะเบียน ' + e.maxCr + ' หน่วยกิต';
    } else if (/^[A-D]\+?$/.test(main.grade)) {
      proposed = 'เลือก'; reason = 'ไม่มีในเล่ม · มีเกรดตัวอักษร → เสนอเป็นวิชาเลือกนอกหลักสูตร';
    } else {
      proposed = 'รอตรวจสอบ'; reason = 'ไม่มีในเล่ม และเกรด ' + (main.grade || 'ว่าง') + ' — ให้เจ้าหน้าที่ระบุหมวด';
    }
    var cat = code in overrides ? overrides[code] : proposed;
    var lowRegs = e.regs.filter(function(r) { return gradeBelowCategory(r.grade, cat); });
    var pendingRegs = e.regs.filter(function(r) { return isPending(r.grade); });
    var eligibleRegs = e.regs.filter(function(r) { return gradeMeetsCategory(r.grade, cat); });
    pass = eligibleRegs.length > 0;
    pending = !pass && pendingRegs.length > 0;
    main = pass ? eligibleRegs[0] : e.regs[0];
    var gradeNeedsReview = !pass && !pending && e.regs.some(function(r) { return isPass(r.grade) && !gradeMeetsCategory(r.grade, cat); });
    var credits = THESIS_CATS.indexOf(cat) >= 0 ? e.maxCr : main.credits;
    var isTransfer = TRANSFER_GRADES.indexOf(main.grade) >= 0;
    var counted = !(b && !courseForPlan(b, gp)) && !(hit && hit.how === 'ambiguous') && pass && !!formRowOf(cat) && NOCREDIT_CATS.indexOf(cat) < 0;
    var review = !!(b && !courseForPlan(b, gp)) || gradeNeedsReview || !b && proposed !== 'ภาษาอังกฤษ' || cat === 'รอตรวจสอบ' || !!(hit && hit.how === 'ambiguous') || !!(b && b.credits != null && b.credits !== credits);
    return {
      code: code, title: e.title, major: b ? b.major || '' : '', nameBook: b ? b.name : '', credits: credits, bookCredits: b ? b.credits : null,
      bookKey: b ? b.key : '', bookDisplay: b ? b.display : '', how: hit ? hit.how : '', candidates: hit && hit.how === 'ambiguous' ? hit.candidates : null,
      grade: main.grade, term: main.term, regs: e.regs, times: e.regs.length,
      inBook: !!b, group: b ? b.group : '', evidence: b ? b.evidence : null, groupEvidence: g ? g.evidence : null,
      proposed: proposed, reason: reason, cat: cat, overridden: code in overrides && overrides[code] !== proposed,
      lowRegs: lowRegs, pendingRegs: pendingRegs, gradeNeedsReview: gradeNeedsReview,
      pass: pass, pending: pending, failed: !pass && !pending, transfer: isTransfer, counted: counted, review: review
    };
  });

  // วิทยานิพนธ์/สารนิพนธ์ของนักศึกษา (ยึดหน่วยกิตที่ลงทะเบียน)
  var thesisC = courses.filter(function (x) { return THESIS_CATS.indexOf(x.cat) >= 0; })[0] || null;
  var thesis = thesisC ? { code: thesisC.code, credits: thesisC.credits, cat: thesisC.cat, times: thesisC.times,
    grades: thesisC.regs.map(function (r) { return r.grade || 'ว่าง'; }) } : null;

  // ตรวจตามหมวดในเล่ม
  var groups = gp.groups.map(function (g) {
    var isThesis = THESIS_CATS.indexOf(g.cat) >= 0;
    var req = g.required;
    return { label: g.label, bookLabel: g.label, cat: g.cat,
      formRow: formRowOf(g.cat), required: req, bookRequired: g.required, atLeast: g.atLeast, evidence: g.evidence,
      noCredit: NOCREDIT_CATS.indexOf(g.cat) >= 0, study: 0, transfer: 0, pending: 0, items: [], missing: [] };
  });
  function assign(x) {
    if (x.inBook && !x.overridden) { var h = groups.filter(function (g) { return g.bookLabel === x.group; })[0]; if (h) return h; }
    return groups.filter(function (g) { return g.formRow && g.formRow === formRowOf(x.cat); })[0] || null;
  }
  courses.forEach(function (x) {
    var g = assign(x);
    if (!g || NOCREDIT_CATS.indexOf(x.cat) >= 0) return;
    g.items.push(x);
    if (x.counted) g[x.transfer ? 'transfer' : 'study'] += x.credits;
    else if (x.pending) g.pending += x.credits;
  });
  curr.courses.forEach(function (b) {
    var g = groups.filter(function (gg) { return gg.bookLabel === b.group; })[0];
    if (!g || g.cat === 'เลือก' || g.noCredit || !courseForPlan(b, gp)) return;
    var choice = (curr.choiceSets || []).filter(function(c) {return c.group === b.group && c.codes.indexOf(b.key)>=0;})[0];
    if (choice) {
      var eligible = courses.filter(function(c) {return choice.codes.indexOf(c.bookKey)>=0 && (c.pass || c.pending);}).length;
      var transferred = choice.codes.some(function(code) {return !!transfer[code];});
      if (eligible >= choice.count) return;

var pdfReady = null;
function getPdfjs() {
  if (pdfReady) return pdfReady;
  pdfReady = new Promise(function (ok, no) {
    var n = 0;
    (function wait() {
      if (window.pdfjsLib) return ok(window.pdfjsLib);
      if (++n > 100) return no(new Error('โหลดตัวอ่าน PDF ไม่สำเร็จ — ลองรีเฟรชหน้า หรือใช้ “วางข้อความแทน”'));
      setTimeout(wait, 100);
    })();
  }).then(function (lib) {
    if (lib.__ready) return lib;
    lib.GlobalWorkerOptions.workerSrc = '/vendor/pdf.worker.min.js?token=' + TOKEN;
    lib.__ready = true;
    return lib;
  });
  pdfReady.catch(function () { pdfReady = null; });
  return pdfReady;
}
function pdfToText(file) {
  var lib;
  return getPdfjs().then(function (l) { lib = l; return file.arrayBuffer(); })
    .then(function (buf) { return lib.getDocument({ data: new Uint8Array(buf), isEvalSupported: false }).promise; })
    .then(function (pdf) {
      var ps = [];
      for (var p = 1; p <= pdf.numPages; p++) ps.push(pdf.getPage(p).then(function (pg) { return pg.getTextContent(); }));
      return Promise.all(ps);
    }).then(function (contents) {
      return contents.map(function (tc, pi) {
        var lines = [];
        tc.items.forEach(function (it, seq) {
          if (!it.str) return;
          var y = it.transform[5], x = it.transform[4];
          var ln = null;
          for (var i = 0; i < lines.length; i++) if (Math.abs(lines[i].y - y) < Math.max(1, Math.min(2.2, (it.height || 10) * 0.18))) { ln = lines[i]; break; }
          if (!ln) { ln = { y: y, items: [] }; lines.push(ln); }
          ln.items.push({ x: x, w: it.width, s: cleanPdfText(it.str), seq: seq });
        });
        lines.sort(function (a, b) { return b.y - a.y; });
        return '@@PAGE ' + (pi + 1) + '@@\n' + lines.map(function (l) {
          l.items.sort(function (a, b) { return a.x - b.x || a.seq - b.seq; });
          var o = '', end = null;
          l.items.forEach(function (it) { if (end !== null && it.x - end > 1.5 && !/\s$/.test(o)) o += ' '; o += it.s; end = it.x + it.w; });
          return cleanPdfText(o).replace(/\s+/g, ' ').trim();
        }).filter(Boolean).join('\n');
      }).join('\n');
    });
}

function hl(line, code) {
  var num = code.slice(-3), d = '๐๑๒๓๔๕๖๗๘๙';
  var th = num.split('').map(function (c) { return '[' + c + d[+c] + ']'; }).join('');
  var re = new RegExp('((?:[A-Z]{3,4}|[ก-ฮ]{2,5})\\s?' + th + ')(?![0-9๐-๙])', 'g');
  return esc(line).replace(re, '<mark>$1</mark>');
}

  if (!S.book || !S.stu) return;
  S.result = evaluate(S.book, S.stu, { overrides: S.overrides, transfer: S.transfer, major: S.major });
  if (S.step === 2) render2();
  if (S.step === 3) render3();
}
function go(n) {
  n = +n;
  if (n > 1 && !(S.book && S.stu)) return;
  if (n > 1 && !majorReady()) { toast('รายวิชาที่พบยังแยกวิชาเอกไม่ได้ กรุณาเลือกวิชาเอก'); return; }
  S.step = n;
  [].forEach.call(document.querySelectorAll('section[data-step]'), function (s) { s.classList.toggle('hidden', +s.dataset.step !== n); });
  [].forEach.call(document.querySelectorAll('.tab'), function (b) { b.setAttribute('aria-selected', +b.dataset.go === n ? 'true' : 'false'); });
  if (n === 2 && S.book && S.stu && $('chkSaveBook').checked && !S.book.savedId) saveBook();
  compute();
  window.scrollTo({ top: 0, behavior: 'smooth' });
}
function saveBook() {
  var key = (S.stu.programKey || '').toUpperCase();
  var c = S.book;
  call('saveCurriculum', { key: key, name: c.name || S.bookName.replace(/\.pdf$/i, ''), data: JSON.stringify(c) })
    .then(function (id) { c.savedId = id; toast('บันทึกเล่มหลักสูตรแล้ว'); loadSavedList(); })
    .catch(function (e) { toast(e.message); });
}

document.addEventListener('change', function (e) {
  var el = e.target;
  if (el.classList.contains('studentField') && S.stu) { S.stu[el.dataset.field] = el.dataset.field === 'plan' ? normalizePlan(el.value) : el.value.trim(); S.result = null; S.savedAt = ''; refresh1(); }
  if (el.classList.contains('ovr')) {
    var x = S.result.courses.filter(function (c) { return c.code === el.dataset.code; })[0];
    if (x && el.value === x.proposed) delete S.overrides[el.dataset.code]; else S.overrides[el.dataset.code] = el.value;
    S.savedAt = ''; compute();
  }
  if (el.classList.contains('trf')) { if (el.checked) S.transfer[el.dataset.code] = true; else delete S.transfer[el.dataset.code]; S.savedAt = ''; compute(); }
  if (el.classList.contains('gmap')) { S.book.groups[+el.dataset.i].cat = el.value; S.book.savedId = S.book.savedId ? '' : ''; refresh1(); }
  if (el.id === 'majorSelect') { S.major = el.value; S.result = null; S.savedAt = ''; refresh1(); autoInspect(); }
  if (el.id === 'filterSel') render2();
  if (el.id === 'savedSel') loadSaved(el.value);
});
